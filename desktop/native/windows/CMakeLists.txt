cmake_minimum_required(VERSION 3.24)
project(codeintel_boundary LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The boundary requires Windows x64 and MSVC")
endif()
add_executable(codeintel-boundary boundary.cpp)
target_compile_features(codeintel-boundary PRIVATE cxx_std_20)
target_compile_definitions(codeintel-boundary PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(codeintel-boundary PRIVATE /W4 /WX /guard:cf /sdl /utf-8)
target_link_options(codeintel-boundary PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf)
target_link_libraries(codeintel-boundary PRIVATE advapi32 bcrypt crypt32)
